**Goal:** Add MFA/2FA support to SSH connections — keyboard-interactive prompts, password+key sequential auth, and built-in TOTP generation.

**Architecture:** SshHandler gains `auth_keyboard_interactive` trait impl that relays prompts to frontend via Tauri events through an mpsc channel. TOTP codes are generated via `totp-rs` crate when a prompt matches keywords and a secret is configured.

```rust
use russh::client::{self, Prompt};
use russh::keys::PublicKey;
use tokio::sync::{mpsc, oneshot};

/// SshHandler 向 session 传递 keyboard-interactive prompt 的请求
pub struct KbPromptRequest {
    pub prompts: Vec<(String, bool)>,
    pub instructions: String,
    pub response_tx: oneshot::Sender<Vec<String>>,
}

pub struct SshHandler {
    /// 用于向 session 传递 keyboard-interactive prompts
    pub kb_tx: Option<mpsc::Sender<KbPromptRequest>>,
}

impl client::Handler for SshHandler {
    type Error = anyhow::Error;

    async fn check_server_key(
        &mut self,
        _server_public_key: &PublicKey,
    ) -> Result<bool, Self::Error> {
        Ok(true)
    }

    async fn auth_keyboard_interactive(
        &mut self,
        _user: &str,
        _submethods: &str,
        _instructions: &str,
        prompts: &[Prompt<'_>],
    ) -> Result<Vec<String>, Self::Error> {
        let (resp_tx, resp_rx) = oneshot::channel();

        let request = KbPromptRequest {
            prompts: prompts
                .iter()
                .map(|p| (p.prompt.to_string(), p.echo))
                .collect(),
            instructions: _instructions.to_string(),
            response_tx: resp_tx,
        };

        if let Some(tx) = &self.kb_tx {
            tx.send(request)
                .await
                .map_err(|_| anyhow::anyhow!("keyboard-interactive channel closed"))?;
        } else {
            anyhow::bail!("keyboard-interactive not configured (no kb_tx channel)");
        }

        match tokio::time::timeout(std::time::Duration::from_secs(60), resp_rx).await {
            Ok(Ok(responses)) => {
                if responses.len() != prompts.len() {
                    anyhow::bail!(
                        "response count mismatch: expected {}, got {}",
                        prompts.len(),
                        responses.len()
                    );
                }
                Ok(responses)
            }
            Ok(Err(_)) => anyhow::bail!("keyboard-interactive response channel closed"),
            Err(_) => anyhow::bail!("keyboard-interactive response timed out (60s)"),
        }
    }
}
```

    fn generate_totp(&self) -> Option<String> {
        let secret = self.config.kb_interactive.as_ref()?.totp_secret.as_ref()?;
        use totp_rs::{Algorithm, TOTP, Secret};
        let secret_bytes = Secret::Encoded(secret.clone()).to_bytes().ok()?;
        let totp = TOTP::new(Algorithm::SHA1, 6, 1, 30, secret_bytes, None, String::new()).ok()?;
        totp.generate_current().ok()

use russh::client::{self, Prompt};
use russh::keys::PublicKey;
use tokio::sync::{mpsc, oneshot};

pub struct KbPromptRequest {
    pub prompts: Vec<(String, bool)>,
    pub instructions: String,
    pub response_tx: oneshot::Sender<Vec<String>>,
}

pub struct SshHandler {
    pub kb_tx: Option<mpsc::Sender<KbPromptRequest>>,
}

impl client::Handler for SshHandler {
    type Error = anyhow::Error;

    async fn check_server_key(
        &mut self,
        _server_public_key: &PublicKey,
    ) -> Result<bool, Self::Error> {
        Ok(true)
    }

    async fn auth_keyboard_interactive(
        &mut self,
        _user: &str,
        _submethods: &str,
        _instructions: &str,
        prompts: &[Prompt<'_>],
    ) -> Result<Vec<String>, Self::Error> {
        let (resp_tx, resp_rx) = oneshot::channel();

        if let Some(tx) = &self.kb_tx {
            tx.send(KbPromptRequest {
                prompts: prompts
                    .iter()
                    .map(|p| (p.prompt.to_string(), p.echo))
                    .collect(),
                instructions: _instructions.to_string(),
                response_tx: resp_tx,
            })
            .await
            .map_err(|_| anyhow::anyhow!("kb prompt channel closed"))?;
        } else {
            anyhow::bail!("keyboard-interactive not configured");
        }

        match tokio::time::timeout(std::time::Duration::from_secs(360), resp_rx).await {
            Ok(Ok(responses)) => {
                if responses.len() != prompts.len() {
                    anyhow::bail!(
                        "response count mismatch: expected {}, got {}",
                        prompts.len(),
                        responses.len()
                    );
                }
                Ok(responses)
            }
            Ok(Err(_)) => anyhow::bail!("keyboard-interactive response channel closed"),
            Err(_) => anyhow::bail!("keyboard-interactive response timed out (360s)"),
        }
    }
}

/// 对已建的 handle 执行认证。供 connect_and_auth 和 jump host 场景复用。
async fn authenticate_with_handle(
    handle: &mut Handle<SshHandler>,
    username: &str,
    auth: &SshAuth,
) -> Result<(), String> {
    match auth {
        SshAuth::Password(password) => {
            let result = handle
                .authenticate_password(username, password.as_str())
                .await
                .map_err(|e| format!("Password auth failed: {}", e))?;
            if !result.success() {
                return Err("Password authentication failed".to_string());
            }
        }
        SshAuth::PrivateKey { key, passphrase } => {
            let key_pair = russh::keys::decode_secret_key(key, passphrase.as_deref())
                .map_err(|e| format!("Failed to parse private key: {}", e))?;
            let key_with_hash = russh::keys::key::PrivateKeyWithHashAlg::new(
                Arc::new(key_pair),
                None,
            );
            let result = handle
                .authenticate_publickey(username, key_with_hash)
                .await
                .map_err(|e| format!("Public key auth failed: {}", e))?;
            if !result.success() {
                return Err("Public key authentication failed".to_string());
            }
        }
        SshAuth::PasswordAndKey { password, key, passphrase } => {
            // 先密钥
            let key_pair = russh::keys::decode_secret_key(key, passphrase.as_deref())
                .map_err(|e| format!("Failed to parse private key: {}", e))?;
            let key_with_hash = russh::keys::key::PrivateKeyWithHashAlg::new(
                Arc::new(key_pair),
                None,
            );
            let result = handle
                .authenticate_publickey(username, key_with_hash)
                .await
                .map_err(|e| format!("Public key auth failed: {}", e))?;
            if !result.success() {
                // 如果服务器配置了 AuthenticationMethods publickey,password，
                // success() 返回 false 表示部分成功，继续密码认证
            }
            // 再密码
            let result = handle
                .authenticate_password(username, password.as_str())
                .await
                .map_err(|e| format!("Password auth failed: {}", e))?;
            if !result.success() {
                return Err("Password+Key authentication failed".to_string());
            }
        }
    }
    Ok(())
}

/// 生成 6 位 TOTP 验证码。失败返回 None（不阻断连接）。
fn generate_totp(secret: &str) -> Option<String> {
    use totp_rs::{Algorithm, TOTP, Secret};
    let secret_bytes = Secret::Encoded(secret.to_string()).to_bytes().ok()?;
    let totp = TOTP::new(Algorithm::SHA1, 6, 1, 30, secret_bytes, None, String::new()).ok()?;
    totp.generate_current().ok()
}

/// 判断 prompt 文本是否匹配 TOTP 相关关键词

        || lower.contains("verification")
        || lower.contains("otp")
        || lower.contains("code")
        || lower.contains("token")
        || lower.contains("one-time")
        || lower.contains("验证")
        || lower.contains("令牌")
        || lower.contains("一次性")
}
